    1. 0.20 M LiNO₃, Neutral
      pH = 7.00
    2. 0.0040 M Ca(OH)₂, Strong base
      pH = 11.90
    3. 0.30 M HCOOH, Weak acid
      pH = 2.14
    4. 0.20 M NaNO₂, Weak base
      pH = 8.28
    5. 1.4 M HClO₄, Strong acid
      pH = -0.15
    6. 0.50 M HCN / 0.20 M NaCN, Buffer
      pH = 8.81
    7. 1.1 M C₆H₅NHCl (pyridinium chloride), Weak acid
      pH = 2.59
    8. 0.40 M HCl / 0.20 M NaCl , Strong acid
      pH = 0.40
